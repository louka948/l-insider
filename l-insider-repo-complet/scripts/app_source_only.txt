function playerInEra(p, era) {
  if (era.id === "ALL") return true;
  return p.e.some((decade) => decade >= era.from && decade < era.to);
}

function playersForCategory(cat, club) {
  if (cat === "GK" || cat === "DEF" || cat === "MID" || cat === "ATT") return PLAYERS.filter((p) => p.p === cat);
  if (cat === "BO") return PLAYERS.filter((p) => p.b);
  if (cat === "LEGEND") return PLAYERS.filter((p) => p.l && p.p !== "COACH");
  if (cat === "COACH") return PLAYERS.filter((p) => p.co);
  if (cat === "CLUB") return club ? PLAYERS.filter((p) => p.cl === club) : [];
  return PLAYERS.filter((p) => p.p !== "COACH");
}

function eraOverlap(e1, e2) {
  return e1.filter((x) => e2.includes(x)).length;
}

function scorePair(a, b) {
  let s = 0;
  if (a.p === b.p) s += 3;
  s += eraOverlap(a.e, b.e);
  if (a.c === b.c) s += 1;
  if (a.b === b.b) s += 0.5;
  return s;
}

// Compose un duo à la volée : on choisit un joueur au hasard dans la
// catégorie, on cherche des partenaires au profil proche, puis on tire
// parmi les meilleurs candidats — jamais toujours le même partenaire.
// On évite aussi de retomber sur un duo déjà tiré cette session.
function pickPair(pool, usedKeys) {
  for (let attempt = 0; attempt < 30; attempt++) {
    const anchor = pool[Math.floor(Math.random() * pool.length)];
    const scored = pool
      .filter((p) => p.n !== anchor.n)
      .map((p) => ({ p, s: scorePair(anchor, p) }))
      .sort((a, b) => b.s - a.s);
    if (scored.length === 0) continue;
    const topN = scored.slice(0, Math.min(8, scored.length));
    const partner = topN[Math.floor(Math.random() * topN.length)].p;
    const key = [anchor.n, partner.n].sort().join("|");
    if (!usedKeys.has(key)) return { a: anchor, b: partner, key };
  }
  const anchor = pool[Math.floor(Math.random() * pool.length)];
  const others = pool.filter((p) => p.n !== anchor.n);
  const partner = others[Math.floor(Math.random() * others.length)];
  return { a: anchor, b: partner, key: [anchor.n, partner.n].sort().join("|") };
}

function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function slugify(name) {
  return name
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/(^-|-$)/g, "");
}

// Avatar de carte : essaie d'abord PLAYER_IMAGES, puis le dossier
// "images/" auto-nommé, sinon retombe sur la silhouette générique.
function PlayerAvatar({ name, size = 84 }) {
  const [stage, setStage] = useState(0); // 0 = image manuelle/auto, 1 = échec -> silhouette
  const meta = PLAYERS.find((p) => p.n === name);
  const pos = meta ? meta.p : "ATT";
  const color = POSITION_COLORS[pos] || "#4B4B4B";
  const imgUrl = PLAYER_IMAGES[name] || `images/${slugify(name)}.jpg`;

  if (stage === 0) {
    return (
      <img
        src={imgUrl}
        alt=""
        onError={() => setStage(1)}
        style={{ width: size, height: size, borderRadius: "50%", objectFit: "cover", objectPosition: "center 15%", border: `2px solid ${color}`, display: "block" }}
      />
    );
  }

  return (
    <svg viewBox="0 0 100 100" width={size} height={size} style={{ display: "block" }}>
      <circle cx="50" cy="50" r="48" fill={color} opacity="0.12" />
      <circle cx="50" cy="33" r="13" fill={color} />
      <path d="M27 93 C27 67 36 57 50 57 C64 57 73 67 73 93 Z" fill={color} />
      <path d="M50 57 L44 68 L50 76 L56 68 Z" fill="#fff" opacity="0.85" />
    </svg>
  );
}

function ImposteurFoot() {
  const [phase, setPhase] = useState("setup"); // setup | reveal | discuss | vote | eliminated | blancGuess | end
  const [count, setCount] = useState(6);
  const [names, setNames] = useState(Array.from({ length: 6 }, (_, i) => `Joueur ${i + 1}`));
  const [photos, setPhotos] = useState(Array.from({ length: 6 }, () => null));
  const [numUndercover, setNumUndercover] = useState(1);
  const [mrBlanc, setMrBlanc] = useState(false);
  const [category, setCategory] = useState("LIBRE");
  const [selectedClub, setSelectedClub] = useState(null);
  const [era, setEra] = useState("ALL");
  const [mode, setMode] = useState("clue");
  const [hardMode, setHardMode] = useState(false);

  const [usedPairKeys, setUsedPairKeys] = useState(new Set());
  const [players, setPlayers] = useState([]);
  const [pairWords, setPairWords] = useState({ civil: "", undercover: "" });
  const [revealIndex, setRevealIndex] = useState(0);
  const [cardFlipped, setCardFlipped] = useState(false);
  const [round, setRound] = useState(1);
  const [selectedVote, setSelectedVote] = useState(null);
  const [lastEliminated, setLastEliminated] = useState(null);
  const [winner, setWinner] = useState(null); // 'civils' | 'infiltres' | 'blanc'
  const [blancGuess, setBlancGuess] = useState("");
  const [blancResult, setBlancResult] = useState(null);
  const [questionTurnIndex, setQuestionTurnIndex] = useState(0);
  const [sampleQuestions, setSampleQuestions] = useState([]);
  const [musicOn, setMusicOn] = useState(false);
  const [showRules, setShowRules] = useState(false);
  const audioRef = useRef({ ctx: null, nodes: [] });

  // Ambiance sonore générée en direct (pas de fichier audio externe donc
  // aucun souci de droits) : nappe de synthé douce + bruit filtré façon
  // rumeur de stade, très discrète.
  function startAmbience() {
    const AC = window.AudioContext || window.webkitAudioContext;
    if (!AC) return;
    const ctx = audioRef.current.ctx || new AC();
    audioRef.current.ctx = ctx;
    const master = ctx.createGain();
    master.gain.value = 0.05;
    master.connect(ctx.destination);

    const freqs = [110, 165, 220];
    const nodes = [];
    freqs.forEach((f, i) => {
      const osc = ctx.createOscillator();
      osc.type = "sine";
      osc.frequency.value = f;
      const g = ctx.createGain();
      g.gain.value = 0.5 / freqs.length;
      osc.connect(g);
      g.connect(master);
      osc.start();
      nodes.push(osc, g);
    });

    const bufferSize = 2 * ctx.sampleRate;
    const noiseBuffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const output = noiseBuffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) output[i] = Math.random() * 2 - 1;
    const noise = ctx.createBufferSource();
    noise.buffer = noiseBuffer;
    noise.loop = true;
    const filter = ctx.createBiquadFilter();
    filter.type = "lowpass";
    filter.frequency.value = 300;
    const noiseGain = ctx.createGain();
    noiseGain.gain.value = 0.4;
    noise.connect(filter);
    filter.connect(noiseGain);
    noiseGain.connect(master);
    noise.start();
    nodes.push(noise, filter, noiseGain, master);

    audioRef.current.nodes = nodes;
  }

  function stopAmbience() {
    audioRef.current.nodes.forEach((n) => {
      try {
        if (n.stop) n.stop();
        n.disconnect();
      } catch (e) {}
    });
    audioRef.current.nodes = [];
  }

  function toggleMusic() {
    if (musicOn) {
      stopAmbience();
      setMusicOn(false);
    } else {
      startAmbience();
      setMusicOn(true);
    }
  }

  const alivePlayers = players.filter((p) => p.alive);

  function resizeNames(n) {
    setCount(n);
    setNames((prev) => {
      const next = [...prev];
      while (next.length < n) next.push(`Joueur ${next.length + 1}`);
      while (next.length > n) next.pop();
      return next;
    });
    setPhotos((prev) => {
      const next = [...prev];
      while (next.length < n) next.push(null);
      while (next.length > n) next.pop();
      return next;
    });
    setNumUndercover((u) => Math.min(u, Math.max(1, Math.floor(n / 3))));
  }

  function enterDiscuss() {
    setQuestionTurnIndex(0);
    const bank = hardMode && mode !== "clue" ? QUESTION_BANK.filter((q) => !q.phys) : QUESTION_BANK;
    setSampleQuestions(shuffle(bank).slice(0, 6).map((q) => q.t));
    setPhase("discuss");
  }

  function startGame() {
    let pool = playersForCategory(category, selectedClub);
    const eraObj = ERA_OPTIONS.find((e) => e.id === era) || ERA_OPTIONS[0];
    pool = pool.filter((p) => playerInEra(p, eraObj));
    if (pool.length < 2) pool = PLAYERS.filter((p) => playerInEra(p, eraObj));
    if (pool.length < 2) pool = PLAYERS;

    const result = pickPair(pool, usedPairKeys);
    const newUsed = new Set(usedPairKeys);
    newUsed.add(result.key);
    const maxCombos = (pool.length * (pool.length - 1)) / 2;
    setUsedPairKeys(newUsed.size >= maxCombos ? new Set() : newUsed);

    const swap = Math.random() < 0.5;
    const civilWord = swap ? result.a.n : result.b.n;
    const undercoverWord = swap ? result.b.n : result.a.n;
    setPairWords({ civil: civilWord, undercover: undercoverWord });

    const roles = [];
    for (let i = 0; i < numUndercover; i++) roles.push("undercover");
    if (mrBlanc) roles.push("blanc");
    while (roles.length < count) roles.push("civil");
    const shuffledRoles = shuffle(roles);

    const built = names.map((name, i) => ({
      id: i,
      name: name.trim() || `Joueur ${i + 1}`,
      photo: photos[i] || null,
      role: shuffledRoles[i],
      word: shuffledRoles[i] === "civil" ? civilWord : shuffledRoles[i] === "undercover" ? undercoverWord : null,
      alive: true,
    }));

    setPlayers(built);
    setRevealIndex(0);
    setCardFlipped(false);
    setRound(1);
    setSelectedVote(null);
    setLastEliminated(null);
    setWinner(null);
    setBlancGuess("");
    setBlancResult(null);
    setPhase("reveal");
  }

  function nextReveal() {
    setCardFlipped(false);
    if (revealIndex + 1 >= players.length) {
      enterDiscuss();
    } else {
      setRevealIndex((i) => i + 1);
    }
  }

  function checkWinner(currentPlayers) {
    const alive = currentPlayers.filter((p) => p.alive);
    const infiltres = alive.filter((p) => p.role === "undercover" || p.role === "blanc").length;
    const civils = alive.filter((p) => p.role === "civil").length;
    if (infiltres === 0) return "civils";
    if (infiltres >= civils) return "infiltres";
    return null;
  }

  function eliminate(id) {
    const target = players.find((p) => p.id === id);
    const updated = players.map((p) => (p.id === id ? { ...p, alive: false } : p));
    setPlayers(updated);
    setLastEliminated(target);
    setSelectedVote(null);

    if (target.role === "blanc") {
      setPhase("blancGuess");
      return;
    }

    const w = checkWinner(updated);
    if (w) {
      setWinner(w);
      setPhase("end");
    } else {
      setPhase("eliminated");
    }
  }

  function submitBlancGuess() {
    const correct = blancGuess.trim().toLowerCase() === pairWords.civil.trim().toLowerCase();
    setBlancResult(correct);
    if (correct) {
      setWinner("blanc");
      setPhase("end");
    } else {
      const w = checkWinner(players);
      if (w) {
        setWinner(w);
        setPhase("end");
      } else {
        setPhase("eliminated");
      }
    }
  }

  function continueRound() {
    setRound((r) => r + 1);
    enterDiscuss();
  }

  function resetGame() {
    setPhase("setup");
    setPlayers([]);
    setWinner(null);
  }

  const currentReveal = players[revealIndex];

  return (
    <div style={styles.page}>
      <style>{FONT_CSS}</style>
      <div style={styles.pitchLines} />
      <div style={styles.floodlight} />

      <div style={styles.shell}>
        <header style={styles.header}>
          <div style={styles.headerTopRow}>
            <button onClick={() => setShowRules(true)} style={styles.iconBtn} aria-label="Règles du jeu">
              <HelpCircle size={18} />
            </button>
            <div style={styles.headerLeft}>
              <Flag size={18} color="#E8B923" strokeWidth={2.5} />
              <span style={styles.headerKicker}>UNDERCOVER</span>
            </div>
            <button onClick={toggleMusic} style={styles.iconBtn} aria-label="Ambiance sonore">
              {musicOn ? <Volume2 size={18} /> : <VolumeX size={18} />}
            </button>
          </div>
          <h1 style={styles.headerTitle}>IMPOSTEUR&nbsp;FOOT</h1>
        </header>

        {showRules && <RulesScreen onClose={() => setShowRules(false)} />}

        {phase === "setup" && (
          <SetupScreen
            count={count}
            resizeNames={resizeNames}
            names={names}
            setNames={setNames}
            photos={photos}
            setPhotos={setPhotos}
            numUndercover={numUndercover}
            setNumUndercover={setNumUndercover}
            mrBlanc={mrBlanc}
            setMrBlanc={setMrBlanc}
            category={category}
            setCategory={setCategory}
            selectedClub={selectedClub}
            setSelectedClub={setSelectedClub}
            era={era}
            setEra={setEra}
            mode={mode}
            setMode={setMode}
            hardMode={hardMode}
            setHardMode={setHardMode}
            onStart={startGame}
          />
        )}

        {phase === "reveal" && currentReveal && (
          <RevealScreen
            key={currentReveal.id}
            player={currentReveal}
            index={revealIndex}
            total={players.length}
            flipped={cardFlipped}
            onFlip={() => setCardFlipped(true)}
            onNext={nextReveal}
          />
        )}

        {phase === "discuss" && mode === "clue" && (
          <ClueScreen round={round} alivePlayers={alivePlayers} onVote={() => setPhase("vote")} />
        )}
        {phase === "discuss" && mode === "turns" && (
          <TurnQuestionScreen
            round={round}
            alivePlayers={alivePlayers}
            turnIndex={questionTurnIndex}
            questions={sampleQuestions}
            hardMode={hardMode}
            onNextTurn={() => setQuestionTurnIndex((i) => i + 1)}
            onVote={() => setPhase("vote")}
          />
        )}
        {phase === "discuss" && mode === "freestyle" && (
          <FreestyleScreen round={round} alivePlayers={alivePlayers} questions={sampleQuestions} hardMode={hardMode} onVote={() => setPhase("vote")} />
        )}

        {phase === "vote" && (
          <VoteScreen
            alivePlayers={alivePlayers}
            selected={selectedVote}
            setSelected={setSelectedVote}
            onConfirm={() => eliminate(selectedVote)}
          />
        )}

        {phase === "eliminated" && lastEliminated && (
          <EliminatedScreen player={lastEliminated} onContinue={continueRound} round={round} />
        )}

        {phase === "blancGuess" && (
          <BlancGuessScreen
            guess={blancGuess}
            setGuess={setBlancGuess}
            onSubmit={submitBlancGuess}
            result={blancResult}
            civilWord={pairWords.civil}
            onContinue={continueRound}
          />
        )}

        {phase === "end" && (
          <EndScreen winner={winner} players={players} pairWords={pairWords} onReset={resetGame} />
        )}
      </div>
    </div>
  );
}

function SetupScreen({ count, resizeNames, names, setNames, photos, setPhotos, numUndercover, setNumUndercover, mrBlanc, setMrBlanc, category, setCategory, selectedClub, setSelectedClub, era, setEra, mode, setMode, hardMode, setHardMode, onStart }) {
  const maxUndercover = Math.max(1, Math.floor(count / 3));
  const noClubSelected = category === "CLUB" && !selectedClub;
  const noRoleSelected = numUndercover === 0 && !mrBlanc;
  const disableStart = noRoleSelected || noClubSelected;
  const [activeIndex, setActiveIndex] = useState(null);
  const fileInputRef = useRef(null);

  function triggerUpload(i) {
    setActiveIndex(i);
    if (fileInputRef.current) fileInputRef.current.click();
  }

  function handleFileChange(e) {
    const file = e.target.files && e.target.files[0];
    e.target.value = "";
    if (!file || activeIndex === null) return;
    const reader = new FileReader();
    reader.onload = () => {
      const next = [...photos];
      next[activeIndex] = reader.result;
      setPhotos(next);
    };
    reader.readAsDataURL(file);

  return (
    <div style={styles.confettiWrap} aria-hidden="true">

            animationDelay: `${p.delay}s`,

          }}
        />
      ))}
    </div>
  );
}

  const colorMap = {
    civils: { win: "#1E7A4C", lose: "#C81E3A" },

    blanc: { win: "#4B4B4B", lose: "#1E7A4C" },
